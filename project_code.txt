package com.ahmed.firstaidhero1.adapter;

import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ImageView;
import android.widget.TextView;

import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.model.CardItem;

import java.util.List;

public class CardAdapter extends RecyclerView.Adapter<CardAdapter.CardViewHolder> {

    public interface OnItemClickListener {
        void onItemClick(CardItem item);
    }

    private final List<CardItem> items;
    private final OnItemClickListener listener;

    public CardAdapter(List<CardItem> items, OnItemClickListener listener) {
        this.items = items;
        this.listener = listener;
    }

    @NonNull
    @Override
    public CardViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_card, parent, false);
        return new CardViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull CardViewHolder holder, int position) {
        CardItem item = items.get(position);
        holder.title.setText(item.getTitle());
        holder.description.setText(item.getDescription());
        holder.image.setImageResource(item.getImageRes());
        holder.severity.setImageResource(item.getSeverity().iconRes);
        holder.severity.setContentDescription(
                holder.itemView.getContext().getString(item.getSeverity().labelRes));
        holder.itemView.setOnClickListener(v -> listener.onItemClick(item));
    }

    @Override
    public int getItemCount() {
        return items.size();
    }

    static class CardViewHolder extends RecyclerView.ViewHolder {
        final ImageView image;
        final ImageView severity;
        final TextView title;
        final TextView description;

        CardViewHolder(@NonNull View itemView) {
            super(itemView);
            image = itemView.findViewById(R.id.cardImage);
            severity = itemView.findViewById(R.id.cardSeverity);
            title = itemView.findViewById(R.id.cardTitle);
            description = itemView.findViewById(R.id.cardDescription);
        }
    }
}

package com.ahmed.firstaidhero1.data;

import android.content.Context;
import android.content.res.Resources;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.model.CardItem;
import com.ahmed.firstaidhero1.model.CaseDetail;
import com.ahmed.firstaidhero1.model.Severity;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class CaseRepository {

    // تعريف حالة واحدة: أرقام الموارد فقط، النصوص الفعلية بتتجاب حسب لغة الجهاز
    private static final class CaseDef {
        final int id;
        final int titleRes;
        final int descRes;
        final Severity severity;
        final int helpRes;
        final int stepsRes;
        final int dontsRes;

        CaseDef(int id, int titleRes, int descRes, Severity severity,
                int helpRes, int stepsRes, int dontsRes) {
            this.id = id;
            this.titleRes = titleRes;
            this.descRes = descRes;
            this.severity = severity;
            this.helpRes = helpRes;
            this.stepsRes = stepsRes;
            this.dontsRes = dontsRes;
        }
    }

    private static final CaseDef[] CASES = {
            new CaseDef(1, R.string.case_bleeding_title, R.string.case_bleeding_desc, Severity.HIGH,
                    R.string.case_bleeding_help, R.array.case_bleeding_steps, R.array.case_bleeding_donts),
            new CaseDef(2, R.string.case_burns_title, R.string.case_burns_desc, Severity.MEDIUM,
                    R.string.case_burns_help, R.array.case_burns_steps, R.array.case_burns_donts),
            new CaseDef(3, R.string.case_cut_title, R.string.case_cut_desc, Severity.LOW,
                    R.string.case_cut_help, R.array.case_cut_steps, R.array.case_cut_donts)
    };

    private final Context context;

    public CaseRepository(Context context) {
        this.context = context;
    }

    // للقائمة: ملخصات فقط
    public List<CardItem> getCardItems() {
        List<CardItem> list = new ArrayList<>();
        for (CaseDef def : CASES) {
            list.add(new CardItem(
                    def.id,
                    context.getString(def.titleRes),
                    context.getString(def.descRes),
                    def.severity,
                    R.mipmap.ic_launcher));
        }
        return list;
    }

    // للتفاصيل: حالة كاملة، أو null إذا الرقم غير موجود
    public CaseDetail getDetail(int id) {
        Resources res = context.getResources();
        for (CaseDef def : CASES) {
            if (def.id == id) {
                return new CaseDetail(
                        def.id,
                        context.getString(def.titleRes),
                        context.getString(def.descRes),
                        def.severity,
                        context.getString(def.helpRes),
                        Arrays.asList(res.getStringArray(def.stepsRes)),
                        Arrays.asList(res.getStringArray(def.dontsRes)));
            }
        }
        return null;
    }
}

package com.ahmed.firstaidhero1.model;

import com.ahmed.firstaidhero1.R;

public enum Severity {
    LOW(1, R.drawable.ic_severity_low, R.string.severity_low),
    MEDIUM(2, R.drawable.ic_severity_medium, R.string.severity_medium),
    HIGH(3, R.drawable.ic_severity_high, R.string.severity_high);

    public final int level;
    public final int iconRes;
    public final int labelRes;

    Severity(int level, int iconRes, int labelRes) {
        this.level = level;
        this.iconRes = iconRes;
        this.labelRes = labelRes;
    }

    // لقراءة القيمة من قاعدة البيانات لاحقاً (عمود INTEGER فيه 1 أو 2 أو 3)
    public static Severity fromLevel(int level) {
        for (Severity s : values()) {
            if (s.level == level) return s;
        }
        return HIGH; // قيمة تالفة أو ناقصة = نفترض الأخطر، مش الأخف
    }
}

package com.ahmed.firstaidhero1.ui;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.widget.ImageView;
import android.widget.LinearLayout;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.appcompat.widget.Toolbar;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowCompat;
import androidx.core.view.WindowInsetsCompat;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.data.CaseRepository;
import com.ahmed.firstaidhero1.model.CaseDetail;
import com.ahmed.firstaidhero1.util.DialHelper;

public class EmergencyDetailActivity extends AppCompatActivity {

    public static final String EXTRA_CASE_ID = "case_id";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_emergency_detail);

        WindowCompat.getInsetsController(getWindow(), getWindow().getDecorView())
                .setAppearanceLightStatusBars(false);

        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.detailRoot), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, 0, systemBars.right, systemBars.bottom);
            return insets;
        });

        // 1) نقرأ الـ id اللي أرسلته الرئيسية ونجيب الحالة
        int caseId = getIntent().getIntExtra(EXTRA_CASE_ID, -1);
        CaseDetail detail = new CaseRepository(this).getDetail(caseId);
        if (detail == null) {
            finish(); // id غير صالح: سكّر الشاشة بدل ما تكرش
            return;
        }

        // 2) الشريط العلوي
        Toolbar toolbar = findViewById(R.id.detailToolbar);
        toolbar.setTitle(detail.getTitle());
        toolbar.setNavigationOnClickListener(v -> finish());

        // 3) الأجزاء الثابتة
        ImageView severityIcon = findViewById(R.id.detailSeverity);
        severityIcon.setImageResource(detail.getSeverity().iconRes);
        severityIcon.setContentDescription(getString(detail.getSeverity().labelRes));

        ((TextView) findViewById(R.id.detailDescription)).setText(detail.getDescription());
        ((TextView) findViewById(R.id.detailHelpText)).setText(detail.getHelpWhen());

        // 4) القوائم: صف لكل خطوة وصف لكل تحذير
        LayoutInflater inflater = LayoutInflater.from(this);

        LinearLayout stepsContainer = findViewById(R.id.stepsContainer);
        for (int i = 0; i < detail.getSteps().size(); i++) {
            View row = inflater.inflate(R.layout.item_step, stepsContainer, false);
            ((TextView) row.findViewById(R.id.stepNumber)).setText(String.valueOf(i + 1));
            ((TextView) row.findViewById(R.id.stepText)).setText(detail.getSteps().get(i));
            stepsContainer.addView(row);
        }

        LinearLayout dontsContainer = findViewById(R.id.dontsContainer);
        for (String text : detail.getDonts()) {
            View row = inflater.inflate(R.layout.item_dont, dontsContainer, false);
            ((TextView) row.findViewById(R.id.dontText)).setText(text);
            dontsContainer.addView(row);
        }

        // 5) زر الاتصال السفلي
        findViewById(R.id.headerButton).setOnClickListener(v -> DialHelper.dialEmergency(this));
    }
}

package com.ahmed.firstaidhero1.ui;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;
import androidx.recyclerview.widget.ConcatAdapter;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.adapter.CardAdapter;
import com.ahmed.firstaidhero1.adapter.HeaderAdapter;
import com.ahmed.firstaidhero1.data.CaseRepository;
import com.ahmed.firstaidhero1.util.DialHelper;

public class HomeFragment extends Fragment {

    public HomeFragment() {
        super(R.layout.fragment_home);
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        RecyclerView recyclerView = view.findViewById(R.id.recyclerView);

        HeaderAdapter headerAdapter = new HeaderAdapter(v ->
                DialHelper.dialEmergency(requireContext()));

        CaseRepository repository = new CaseRepository(requireContext());
        CardAdapter cardAdapter = new CardAdapter(repository.getCardItems(), item -> {
            Intent intent = new Intent(requireContext(), EmergencyDetailActivity.class);
            intent.putExtra(EmergencyDetailActivity.EXTRA_CASE_ID, item.getId());
            startActivity(intent);
        });

        ConcatAdapter concatAdapter = new ConcatAdapter(headerAdapter, cardAdapter);

        recyclerView.setLayoutManager(new LinearLayoutManager(requireContext()));
        recyclerView.setHasFixedSize(true);
        recyclerView.setAdapter(concatAdapter);
    }
}

package com.ahmed.firstaidhero1.ui;

import android.graphics.Color;
import android.os.Bundle;
import android.transition.TransitionManager;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ImageView;

import androidx.activity.EdgeToEdge;
import androidx.activity.OnBackPressedCallback;
import androidx.appcompat.app.AppCompatActivity;
import androidx.appcompat.widget.SearchView;
import androidx.appcompat.widget.Toolbar;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowCompat;
import androidx.core.view.WindowInsetsCompat;
import androidx.fragment.app.Fragment;
import androidx.fragment.app.FragmentManager;
import androidx.fragment.app.FragmentTransaction;

import com.ahmed.firstaidhero1.R;
import com.google.android.material.bottomnavigation.BottomNavigationView;

public class MainActivity extends AppCompatActivity {

    private static final String TAG_HOME = "home";
    private static final String TAG_CONTACTS = "contacts";
    private static final String TAG_TREATED = "treated";
    private static final String TAG_STATS = "stats";
    private static final String[] TAGS = {TAG_HOME, TAG_CONTACTS, TAG_TREATED, TAG_STATS};

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        BottomNavigationView bottomNav = findViewById(R.id.bottomNav);

        WindowCompat.getInsetsController(getWindow(), getWindow().getDecorView())
                .setAppearanceLightStatusBars(false);

        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            Insets ime = insets.getInsets(WindowInsetsCompat.Type.ime());
            // الأعلى: مسؤولية AppBarLayout. الأسفل: الشريط السفلي بيتعامل مع شريط النظام لحاله،
            // فهون نحسب بس الكيبورد
            v.setPadding(systemBars.left, 0, systemBars.right, ime.bottom);
            // الكيبورد مفتوح = نخفي الشريط السفلي حتى ما يطلع فوق الكيبورد
            bottomNav.setVisibility(
                    insets.isVisible(WindowInsetsCompat.Type.ime()) ? View.GONE : View.VISIBLE);
            return insets;
        });

        // ---------- الشريط العلوي (نفس كودك السابق) ----------
        Toolbar toolbar = findViewById(R.id.toolbar);
        setSupportActionBar(toolbar);
        getSupportActionBar().setDisplayShowTitleEnabled(false);

        ImageView logo = findViewById(R.id.logo);
        ViewGroup toolbarContent = findViewById(R.id.toolbarContent);
        SearchView searchView = findViewById(R.id.searchView);

        View plate = searchView.findViewById(androidx.appcompat.R.id.search_plate);
        plate.setBackgroundColor(Color.TRANSPARENT);

        OnBackPressedCallback backCallback = new OnBackPressedCallback(false) {
            @Override
            public void handleOnBackPressed() {
                searchView.clearFocus();
            }
        };
        getOnBackPressedDispatcher().addCallback(this, backCallback);

        searchView.setOnQueryTextFocusChangeListener((v, hasFocus) -> {
            TransitionManager.beginDelayedTransition(toolbarContent);
            logo.setVisibility(hasFocus ? View.GONE : View.VISIBLE);
            backCallback.setEnabled(hasFocus);
        });

        searchView.setOnQueryTextListener(new SearchView.OnQueryTextListener() {
            @Override
            public boolean onQueryTextSubmit(String query) {
                return true;
            }

            @Override
            public boolean onQueryTextChange(String newText) {
                return true;
            }
        });

        // ---------- التنقل بين التبويبات ----------
        setupFragments(savedInstanceState);

        bottomNav.setOnItemSelectedListener(item -> {
            int id = item.getItemId();
            if (id == R.id.nav_home) {
                showTab(TAG_HOME);
            } else if (id == R.id.nav_contacts) {
                showTab(TAG_CONTACTS);
            } else if (id == R.id.nav_treated) {
                showTab(TAG_TREATED);
            } else if (id == R.id.nav_stats) {
                showTab(TAG_STATS);
            } else {
                return false;
            }
            return true;
        });
    }

    private void setupFragments(Bundle savedInstanceState) {
        // لو الشاشة انعادت (تدوير مثلاً)، الـ FragmentManager رجّع الـ Fragments لحاله
        // ولو أضفناهم مرة تانية بيصيروا نسختين فوق بعض
        if (savedInstanceState != null) return;

        Fragment contacts = new ContactsFragment();
        Fragment treated = new TreatedCasesFragment();
        Fragment stats = new StatisticsFragment();

        getSupportFragmentManager().beginTransaction()
                .setReorderingAllowed(true)
                .add(R.id.fragmentContainer, new HomeFragment(), TAG_HOME)
                .add(R.id.fragmentContainer, contacts, TAG_CONTACTS).hide(contacts)
                .add(R.id.fragmentContainer, treated, TAG_TREATED).hide(treated)
                .add(R.id.fragmentContainer, stats, TAG_STATS).hide(stats)
                .commit();
    }

    private void showTab(String selectedTag) {
        FragmentManager fm = getSupportFragmentManager();
        FragmentTransaction ft = fm.beginTransaction().setReorderingAllowed(true);
        for (String tag : TAGS) {
            Fragment f = fm.findFragmentByTag(tag);
            if (f == null) continue;
            if (tag.equals(selectedTag)) ft.show(f);
            else ft.hide(f);
        }
        ft.commit();
    }
}

package com.ahmed.firstaidhero1.util;

import android.content.ActivityNotFoundException;
import android.content.Context;
import android.content.Intent;
import android.net.Uri;
import android.widget.Toast;

import com.ahmed.firstaidhero1.R;

public final class DialHelper {

    private DialHelper() {}

    public static void dialEmergency(Context context) {
        String number = context.getString(R.string.emergency_number);
        Intent intent = new Intent(Intent.ACTION_DIAL, Uri.parse("tel:" + number));
        try {
            context.startActivity(intent);
        } catch (ActivityNotFoundException e) {
            Toast.makeText(context, R.string.no_dialer, Toast.LENGTH_SHORT).show();
        }
    }
}
