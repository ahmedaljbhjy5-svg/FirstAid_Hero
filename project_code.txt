    public interface OnItemClickListener {
        void onItemClick(CardItem item);
    }

    // بتحدد شو تغيّر بين قائمتين، فالتحديث بيحرّك بس الصفوف المتأثرة
    private static final DiffUtil.ItemCallback<CardItem> DIFF = new DiffUtil.ItemCallback<CardItem>() {
        @Override
        public boolean areItemsTheSame(@NonNull CardItem a, @NonNull CardItem b) {
            return a.getId() == b.getId();
        }

        @Override
        public boolean areContentsTheSame(@NonNull CardItem a, @NonNull CardItem b) {
            return a.getTitle().equals(b.getTitle())
                    && a.getDescription().equals(b.getDescription())
                    && a.getSeverity() == b.getSeverity();
        }
    };

    private final OnItemClickListener listener;

    public CardAdapter(OnItemClickListener listener) {
        super(DIFF);
        this.listener = listener;
    }

    @NonNull
    @Override
    public CardViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_card, parent, false);
        return new CardViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull CardViewHolder holder, int position) {
        CardItem item = getItem(position);
        holder.title.setText(item.getTitle());
        holder.description.setText(item.getDescription());
        holder.image.setImageResource(item.getImageRes());

        holder.itemView.setOnClickListener(v -> listener.onItemClick(item));

        final ImageView image;
        final ImageView severity;
        final TextView title;
        final TextView description;

            severity = itemView.findViewById(R.id.cardSeverity);
            title = itemView.findViewById(R.id.cardTitle);

        void onCall(Contact contact);
        void onEdit(Contact contact);
        void onDelete(Contact contact);

    private final Listener listener;

    public ContactAdapter(Listener listener) {
        this.listener = listener;
    }

        items.clear();
        items.addAll(newItems);
        notifyDataSetChanged();
    }

                .inflate(R.layout.item_contact, parent, false);

package com.ahmed.firstaidhero1.adapter;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.time.format.FormatStyle;
import java.util.Objects;

public class TreatedCaseAdapter
        extends ListAdapter<TreatedCase, TreatedCaseAdapter.TreatedViewHolder> {

    public interface OnDeleteListener {
        void onDelete(TreatedCase treatedCase);
    }

    private static final DiffUtil.ItemCallback<TreatedCase> DIFF =
            new DiffUtil.ItemCallback<TreatedCase>() {
                @Override
                public boolean areItemsTheSame(@NonNull TreatedCase a, @NonNull TreatedCase b) {
                    return a.getId() == b.getId();
                }

                @Override
                public boolean areContentsTheSame(@NonNull TreatedCase a, @NonNull TreatedCase b) {
                    return a.getCaseId() == b.getCaseId()
                            && a.getCaseTitle().equals(b.getCaseTitle())
                            && a.getSeverity() == b.getSeverity()
                            && a.getDescription().equals(b.getDescription())
                            && a.getProcedures().equals(b.getProcedures())
                            && a.getDate().equals(b.getDate())
                            && Objects.equals(a.getPhotoPath(), b.getPhotoPath());
                }
            };

    private final OnDeleteListener listener;

    public TreatedCaseAdapter(OnDeleteListener listener) {
        super(DIFF);
        this.listener = listener;
    }

    @NonNull
    @Override
    public TreatedViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_treated_case, parent, false);
        return new TreatedViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull TreatedViewHolder holder, int position) {
        TreatedCase item = getItem(position);
        holder.date.setText(formatDate(item.getDate()));
        holder.caseTitle.setText(item.getCaseTitle());
        holder.severity.setImageResource(item.getSeverity().iconRes);
        holder.description.setText(item.getDescription());
        holder.procedures.setText(item.getProcedures());

        // لازم نعالج الحالتين (بصورة وبدون) بكل bind بسبب إعادة التدوير
        if (item.getPhotoPath() == null) {
            Glide.with(holder.photo).clear(holder.photo);
            holder.photo.setImageResource(R.drawable.ic_photo_placeholder);
        } else {
            Glide.with(holder.photo)
                    .load(PhotoStorage.resolve(holder.itemView.getContext(), item.getPhotoPath()))
                    .centerCrop()
                    .placeholder(R.drawable.ic_photo_placeholder)
                    .error(R.drawable.ic_photo_placeholder) // الملف انحذف أو تالف
                    .into(holder.photo);
        }

        holder.delete.setOnClickListener(v -> listener.onDelete(item));
    }

    private static String formatDate(String iso) {
        try {
            return LocalDate.parse(iso).format(DateTimeFormatter.ofLocalizedDate(FormatStyle.MEDIUM));
        } catch (DateTimeParseException e) {
            return iso; // صيغة غير متوقعة: نعرضها كما هي بدل ما ننهار
        }
    }

    static class TreatedViewHolder extends RecyclerView.ViewHolder {
        final ImageView photo;
        final TextView date;
        final ImageView severity;
        final TextView caseTitle;
        final TextView description;
        final TextView procedures;
        final ImageButton delete;

        TreatedViewHolder(@NonNull View itemView) {
            super(itemView);
            photo = itemView.findViewById(R.id.treatedPhoto);
            date = itemView.findViewById(R.id.treatedDate);

package com.ahmed.firstaidhero1.data;

import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteException;
import android.database.sqlite.SQLiteOpenHelper;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;

public class AppDatabase extends SQLiteOpenHelper {

    private static final String DB_NAME = "firstaid.db";
    // لازم تطابق PRAGMA user_version داخل firstaid.db (ينتجها build_db.py)
    private static final int DB_VERSION = 2;

    // جدول الحالات
    public static final String TABLE_EMERGENCIES = "emergencies";
    public static final String COL_CASE_ID = "case_id";
    public static final String COL_TITLE = "title";
    public static final String COL_DANGER = "danger_level";
    public static final String COL_SYMPTOMS = "symptoms";
    public static final String COL_SEARCH = "search_text";
    public static final String COL_TITLE_EN = "title_en";
    public static final String COL_SYMPTOMS_EN = "symptoms_en";
    public static final String COL_SEARCH_EN = "search_text_en";
    public static final String COL_SOURCE = "source";
    public static final String COL_SOURCE_EN = "source_en";

    // جدول الخطوات
    public static final String TABLE_STEPS = "steps";
    public static final String COL_STEP_NUMBER = "step_number";
    public static final String COL_INSTRUCTION = "instruction_text";
    public static final String COL_INSTRUCTION_EN = "instruction_text_en";

    // جدول الأرقام المخصصة
    public static final String TABLE_CONTACTS = "custom_contacts";
    public static final String COL_ID = "contact_id";
    public static final String COL_NAME = "contact_name";
    public static final String COL_PHONE = "phone_number";

    // جدول الحالات المعالجة
    public static final String TABLE_TREATED = "treated_cases";
    public static final String COL_TREATED_ID = "treated_id";
    public static final String COL_DESCRIPTION = "description";
    public static final String COL_TREATED_DATE = "treated_date";
    public static final String COL_PHOTO = "photo_path";
    public static final String COL_PROCEDURES = "procedures";

    private static AppDatabase instance;

    public static synchronized AppDatabase getInstance(Context context) {
        if (instance == null) {
            Context app = context.getApplicationContext();
            copyFromAssetsIfNeeded(app);
            instance = new AppDatabase(app);
        }
        return instance;
    }

    private AppDatabase(Context context) {
        super(context, DB_NAME, null, DB_VERSION);
    }

    // بتنسخ القاعدة الجاهزة من assets: أول تشغيل، أو إذا الملف الموجود قديم/تالف.
    // تنبيه: استبدال ملف قديم بيحذف الحالات المعالجة المخزّنة فيه (مقبول أثناء التطوير،
    // وللتسليم النهائي بنثبّت DB_VERSION ولا نغيّره)
    private static void copyFromAssetsIfNeeded(Context context) {
        File dbFile = context.getDatabasePath(DB_NAME);
        if (dbFile.exists()) {
            if (isCurrent(dbFile)) return;
            context.deleteDatabase(DB_NAME);
        }

        File dir = dbFile.getParentFile();
        if (dir != null && !dir.exists() && !dir.mkdirs()) {
            throw new IllegalStateException("Cannot create database directory: " + dir);
        }

        // ننسخ لملف مؤقت أولاً، وبعدها نعيد تسميته. لو انقطع النسخ بالنص ما بيضل ملف تالف
        File tmp = new File(dir, DB_NAME + ".tmp");
        try (InputStream in = context.getAssets().open(DB_NAME);
             OutputStream out = new FileOutputStream(tmp)) {
            byte[] buffer = new byte[8192];
            int read;
            while ((read = in.read(buffer)) != -1) {
                out.write(buffer, 0, read);
            }
            out.flush();
        } catch (IOException e) {
            tmp.delete();
            throw new IllegalStateException("Failed to copy " + DB_NAME + " from assets", e);
        }

        if (!tmp.renameTo(dbFile)) {
            tmp.delete();
            throw new IllegalStateException("Failed to install " + DB_NAME);
        }
    }

    // الملف الموجود صالح إذا إصداره مطابق وفيه الأعمدة الأساسية (بدونها ما بتشتغل الاستعلامات)
    private static boolean isCurrent(File dbFile) {
        try (SQLiteDatabase db = SQLiteDatabase.openDatabase(
                dbFile.getPath(), null, SQLiteDatabase.OPEN_READONLY)) {
            return db.getVersion() == DB_VERSION
                    && hasColumn(db, TABLE_EMERGENCIES, COL_TITLE_EN)
                    && hasColumn(db, TABLE_STEPS, COL_INSTRUCTION_EN)
                    && hasColumn(db, TABLE_TREATED, COL_CASE_ID);
        } catch (SQLiteException e) {
            return false; // ملف تالف
        }
    }

    private static boolean hasColumn(SQLiteDatabase db, String table, String column) {
        try (Cursor c = db.rawQuery("PRAGMA table_info(" + table + ")", null)) {
            int nameCol = c.getColumnIndexOrThrow("name");
            while (c.moveToNext()) {
                if (column.equals(c.getString(nameCol))) return true;
            }
        }
        return false;
    }

    @Override
    public void onConfigure(SQLiteDatabase db) {
        super.onConfigure(db);
        db.setForeignKeyConstraintsEnabled(true); // SQLite بيتجاهل المفاتيح الأجنبية ما لم نفعّلها
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        // فاضية عن قصد: الجداول جاهزة داخل الملف المنسوخ (user_version = 1)
    }

    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
        // لاحقاً: ترفع DB_VERSION وتحط هون ALTER TABLE أو CREATE TABLE للتغييرات
    }
}

package com.ahmed.firstaidhero1.data;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.model.Contact;
import com.ahmed.firstaidhero1.util.PhoneValidator;

import java.util.ArrayList;
import java.util.List;

public class ContactRepository {

    private final Context context;
    private final AppDatabase helper;

    public ContactRepository(Context context) {
        this.context = context;
        this.helper = AppDatabase.getInstance(context);
    }

    // الأرقام العامة الثابتة أولاً (id سالب = غير مخزّنة بالقاعدة)، وبعدها أرقام المستخدم بالاسم
    public List<Contact> getAll() {
        List<Contact> list = new ArrayList<>();
        list.add(new Contact(-1, context.getString(R.string.ambulance_label),
                context.getString(R.string.number_ambulance), true));
        list.add(new Contact(-2, context.getString(R.string.fire_label),
                context.getString(R.string.number_fire), true));
        list.add(new Contact(-3, context.getString(R.string.police_label),
                context.getString(R.string.number_police), true));

        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.query(
                AppDatabase.TABLE_CONTACTS,
                null, null, null, null, null,
                AppDatabase.COL_NAME + " COLLATE NOCASE ASC")) {

            int idCol = c.getColumnIndexOrThrow(AppDatabase.COL_ID);
            int nameCol = c.getColumnIndexOrThrow(AppDatabase.COL_NAME);
            int phoneCol = c.getColumnIndexOrThrow(AppDatabase.COL_PHONE);

            while (c.moveToNext()) {
                list.add(new Contact(
                        c.getLong(idCol),
                        c.getString(nameCol),
                        c.getString(phoneCol),
                        false));
            }
        }
        return list;
    }

    public long insert(String name, String phone) {
        return helper.getWritableDatabase()
                .insert(AppDatabase.TABLE_CONTACTS, null, values(name, phone));
    }

    // بترجع عدد الصفوف المعدّلة (0 = الصف انحذف قبل التعديل)
    public int update(long id, String name, String phone) {
        return helper.getWritableDatabase().update(
                AppDatabase.TABLE_CONTACTS,
                values(name, phone),
                AppDatabase.COL_ID + " = ?",
                new String[]{String.valueOf(id)});
    }

    public void delete(long id) {
        helper.getWritableDatabase().delete(
                AppDatabase.TABLE_CONTACTS,
                AppDatabase.COL_ID + " = ?",
                new String[]{String.valueOf(id)});
    }

    // هل الرقم محفوظ مسبقاً؟ excludeId = الصف اللي عم نعدّله (حتى ما يُعتبر مكرراً مع نفسه)
    public boolean phoneExists(String phone, long excludeId) {
        String target = PhoneValidator.canonical(phone);
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.query(
                AppDatabase.TABLE_CONTACTS,
                new String[]{AppDatabase.COL_ID, AppDatabase.COL_PHONE},
                null, null, null, null, null)) {

            int idCol = c.getColumnIndexOrThrow(AppDatabase.COL_ID);
            int phoneCol = c.getColumnIndexOrThrow(AppDatabase.COL_PHONE);
            while (c.moveToNext()) {
                if (c.getLong(idCol) != excludeId
                        && PhoneValidator.canonical(c.getString(phoneCol)).equals(target)) {
                    return true;
                }
            }
        }
        return false;
    }

    private static ContentValues values(String name, String phone) {
        ContentValues values = new ContentValues();
        values.put(AppDatabase.COL_NAME, name);
        values.put(AppDatabase.COL_PHONE, phone);
        return values;
    }
}

package com.ahmed.firstaidhero1.data;

import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.model.CardItem;
import com.ahmed.firstaidhero1.model.CaseDetail;
import com.ahmed.firstaidhero1.model.CaseOption;
import com.ahmed.firstaidhero1.model.Severity;
import com.ahmed.firstaidhero1.util.Lang;
import com.ahmed.firstaidhero1.util.TextNormalizer;

import java.util.ArrayList;
import java.util.List;

public class EmergencyRepository {

    // الحرج أولاً، وداخل كل مستوى بترتيب إدخال الحالات (الأكثر شيوعاً أولاً)
    private static final String DEFAULT_ORDER =
            AppDatabase.COL_DANGER + " DESC, " + AppDatabase.COL_CASE_ID + " ASC";

    // أسماء مستعارة للأعمدة بلغة الجهاز (انظر Lang.pick)
    private static final String ALIAS_TITLE = "title_l";
    private static final String ALIAS_SYMPTOMS = "symptoms_l";
    private static final String ALIAS_STEP = "step_l";
    private static final String ALIAS_SOURCE = "source_l";

    private final AppDatabase helper;

    public EmergencyRepository(Context context) {
        this.helper = AppDatabase.getInstance(context);
    }

    private static String titleColumn() {
        return Lang.pick(AppDatabase.COL_TITLE, AppDatabase.COL_TITLE_EN) + " AS " + ALIAS_TITLE;
    }

    private static String symptomsColumn() {
        return Lang.pick(AppDatabase.COL_SYMPTOMS, AppDatabase.COL_SYMPTOMS_EN) + " AS " + ALIAS_SYMPTOMS;
    }

    // بحث فوري: كل كلمة بالاستعلام لازم تكون موجودة (AND) بالنص العربي أو الإنجليزي،
    // فالمستخدم بيلاقي الحالة مهما كانت لغة واجهته. والنص فاضي = كل الحالات
    public List<CardItem> search(String rawQuery) {
        StringBuilder where = new StringBuilder();
        List<String> args = new ArrayList<>();
        for (String token : TextNormalizer.normalize(rawQuery).split(" ")) {
            if (token.isEmpty()) continue;
            if (where.length() > 0) where.append(" AND ");
            where.append("(")
                    .append(AppDatabase.COL_SEARCH).append(" LIKE ? ESCAPE '\\' OR ")
                    .append(AppDatabase.COL_SEARCH_EN).append(" LIKE ? ESCAPE '\\')");
            String like = "%" + TextNormalizer.escapeLike(token) + "%";
            args.add(like);
            args.add(like);
        }

        List<CardItem> results = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.query(
                AppDatabase.TABLE_EMERGENCIES,
                new String[]{AppDatabase.COL_CASE_ID, titleColumn(),
                        AppDatabase.COL_DANGER, symptomsColumn()},
                where.length() == 0 ? null : where.toString(),
                args.isEmpty() ? null : args.toArray(new String[0]),
                null, null, DEFAULT_ORDER)) {

            int idCol = c.getColumnIndexOrThrow(AppDatabase.COL_CASE_ID);
            int titleCol = c.getColumnIndexOrThrow(ALIAS_TITLE);
            int dangerCol = c.getColumnIndexOrThrow(AppDatabase.COL_DANGER);
            int symptomsCol = c.getColumnIndexOrThrow(ALIAS_SYMPTOMS);

            while (c.moveToNext()) {
                results.add(new CardItem(
                        c.getLong(idCol),
                        c.getString(titleCol),
                        c.getString(symptomsCol),
                        Severity.fromLevel(c.getInt(dangerCol)),
                        R.mipmap.ic_launcher)); // مؤقت لحد ما نجهّز الصور
            }
        }
        return results;
    }

    // كل الحالات (اسم + خطورة) لقائمة الاختيار بشاشة إضافة حالة معالجة، مرتبة أبجدياً
    public List<CaseOption> getCaseOptions() {
        List<CaseOption> options = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.query(
                AppDatabase.TABLE_EMERGENCIES,
                new String[]{AppDatabase.COL_CASE_ID, titleColumn(), AppDatabase.COL_DANGER},
                null, null, null, null,
                ALIAS_TITLE + " COLLATE NOCASE ASC")) {

            int idCol = c.getColumnIndexOrThrow(AppDatabase.COL_CASE_ID);
            int titleCol = c.getColumnIndexOrThrow(ALIAS_TITLE);
            int dangerCol = c.getColumnIndexOrThrow(AppDatabase.COL_DANGER);
            while (c.moveToNext()) {
                options.add(new CaseOption(
                        c.getLong(idCol),
                        c.getString(titleCol),
                        Severity.fromLevel(c.getInt(dangerCol))));
            }
        }
        return options;
    }

    // حالة كاملة مع خطواتها ومصدرها مرتبة، أو null إذا الرقم غير موجود
    public CaseDetail getDetail(long caseId) {
        SQLiteDatabase db = helper.getReadableDatabase();
        String[] idArg = {String.valueOf(caseId)};

        try (Cursor c = db.query(
                AppDatabase.TABLE_EMERGENCIES,
                new String[]{titleColumn(), AppDatabase.COL_DANGER, symptomsColumn(),
                        Lang.pick(AppDatabase.COL_SOURCE, AppDatabase.COL_SOURCE_EN)
                                + " AS " + ALIAS_SOURCE},
                AppDatabase.COL_CASE_ID + " = ?", idArg, null, null, null)) {

            if (!c.moveToFirst()) return null;

            String title = c.getString(c.getColumnIndexOrThrow(ALIAS_TITLE));
            Severity severity = Severity.fromLevel(c.getInt(c.getColumnIndexOrThrow(AppDatabase.COL_DANGER)));
            String symptoms = c.getString(c.getColumnIndexOrThrow(ALIAS_SYMPTOMS));
            String source = c.getString(c.getColumnIndexOrThrow(ALIAS_SOURCE));

            List<String> steps = new ArrayList<>();
            try (Cursor s = db.query(
                    AppDatabase.TABLE_STEPS,
                    new String[]{Lang.pick(AppDatabase.COL_INSTRUCTION, AppDatabase.COL_INSTRUCTION_EN)
                            + " AS " + ALIAS_STEP},
                    AppDatabase.COL_CASE_ID + " = ?", idArg,
                    null, null, AppDatabase.COL_STEP_NUMBER + " ASC")) {
                int textCol = s.getColumnIndexOrThrow(ALIAS_STEP);
                while (s.moveToNext()) {
                    steps.add(s.getString(textCol));
                }
            }
            return new CaseDetail(caseId, title, severity, symptoms, steps, source);
        }
    }
}

package com.ahmed.firstaidhero1.data;

import android.content.Context;

import java.util.ArrayList;
import java.util.List;

// كل الإحصائيات هون مبنية على الحالات المعالجة اللي سجّلناها (treated_cases)،
// مش على حالات الدليل. وخطورة واسم كل حالة بيجوا من JOIN مع emergencies
public class StatsRepository {

    private static final String T = AppDatabase.TABLE_TREATED;
    private static final String E = AppDatabase.TABLE_EMERGENCIES;
    private static final String JOIN = " FROM " + T + " JOIN " + E
            + " ON " + E + "." + AppDatabase.COL_CASE_ID + " = " + T + "." + AppDatabase.COL_CASE_ID;

    private final AppDatabase helper;

    public StatsRepository(Context context) {
        this.helper = AppDatabase.getInstance(context);
    }

    // عدد الحالات المعالجة بكل مستوى خطورة. الحرجة أولاً
    public List<SeverityStat> getSeverityStats() {
        // GROUP BY ما بيرجّع المجموعات الفاضية، فبنبدأ بأصفار ونعبّي اللي رجع
        int[] counts = new int[Severity.values().length + 1]; // الفهرس = رقم المستوى (1 إلى 3)

        String sql = "SELECT " + E + "." + AppDatabase.COL_DANGER + ", COUNT(*)" + JOIN
                + " GROUP BY " + E + "." + AppDatabase.COL_DANGER;
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, null)) {
            while (c.moveToNext()) {
                int level = c.getInt(0);
                if (level >= 1 && level < counts.length) counts[level] = c.getInt(1);
            }
        }

        List<SeverityStat> result = new ArrayList<>();
        result.add(new SeverityStat(Severity.HIGH, counts[Severity.HIGH.level]));
        result.add(new SeverityStat(Severity.MEDIUM, counts[Severity.MEDIUM.level]));
        result.add(new SeverityStat(Severity.LOW, counts[Severity.LOW.level]));
        return result;
    }

    // عدد الحالات المعالجة لكل حالة من الدليل (الأكثر تكراراً أولاً)، للحالات اللي إلها سجل فقط
    public List<CaseStat> getCaseStats() {
        String titleExpr = Lang.pick(E + "." + AppDatabase.COL_TITLE, E + "." + AppDatabase.COL_TITLE_EN);
        String sql = "SELECT " + titleExpr + " AS case_title, "
                + E + "." + AppDatabase.COL_DANGER + ", COUNT(*) AS n" + JOIN
                + " GROUP BY " + T + "." + AppDatabase.COL_CASE_ID
                + " ORDER BY n DESC, " + E + "." + AppDatabase.COL_DANGER + " DESC, "
                + E + "." + AppDatabase.COL_CASE_ID + " ASC";
        List<CaseStat> result = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, null)) {
            while (c.moveToNext()) {
                result.add(new CaseStat(c.getString(0), Severity.fromLevel(c.getInt(1)), c.getInt(2)));
            }
        }
        return result;
    }

    // عدد الحالات المعالجة بكل شهر (yyyy-MM). الأحدث أولاً
    public List<MonthStat> getTreatedPerMonth() {
        String sql = "SELECT substr(" + AppDatabase.COL_TREATED_DATE + ", 1, 7) AS month, COUNT(*)"
                + " FROM " + T
                + " GROUP BY month ORDER BY month DESC";
        List<MonthStat> result = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, null)) {
            while (c.moveToNext()) {
                result.add(new MonthStat(c.getString(0), c.getInt(1)));
            }
        }
        return result;
    }

    // عدد الحالات المعالجة بكل يوم (yyyy-MM-dd)، آخر limit أيام فيها تسجيل، الأحدث أولاً
    public List<DayStat> getTreatedPerDay(int limit) {
        // limit رقم int من عندنا (مش من المستخدم) فآمن نضمّنه بالنص
        String sql = "SELECT " + AppDatabase.COL_TREATED_DATE + ", COUNT(*)"
                + " FROM " + T
                + " GROUP BY " + AppDatabase.COL_TREATED_DATE
                + " ORDER BY " + AppDatabase.COL_TREATED_DATE + " DESC LIMIT " + Math.max(limit, 1);
        List<DayStat> result = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, null)) {
            while (c.moveToNext()) {
                result.add(new DayStat(c.getString(0), c.getInt(1)));
            }
        }
        return result;
    }

    public int getTreatedTotal() {
        String sql = "SELECT COUNT(*) FROM " + T;
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, null)) {
            return c.moveToFirst() ? c.getInt(0) : 0;
        }
    }

package com.ahmed.firstaidhero1.data;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteException;

import androidx.annotation.Nullable;

import com.ahmed.firstaidhero1.model.Severity;
import com.ahmed.firstaidhero1.model.TreatedCase;
import com.ahmed.firstaidhero1.util.Lang;
import com.ahmed.firstaidhero1.util.TextNormalizer;

import java.util.ArrayList;
import java.util.List;

public class TreatedCaseRepository {

    private final AppDatabase helper;

    public TreatedCaseRepository(Context context) {
        this.helper = AppDatabase.getInstance(context);
    }

    // بترمي استثناء إذا فشل الإدخال، حتى يعرف المستدعي ويعرض رسالة بدل ما يظن إنه نجح
    // caseId لازم يكون من جدول emergencies (المفتاح الأجنبي بيرفض غير هيك)
    public long insert(long caseId, String description, String procedures, String dateIso,
                       @Nullable String photoPath) {
        ContentValues values = new ContentValues();
        values.put(AppDatabase.COL_CASE_ID, caseId);
        values.put(AppDatabase.COL_DESCRIPTION, description);
        values.put(AppDatabase.COL_PROCEDURES, procedures);
        values.put(AppDatabase.COL_TREATED_DATE, dateIso);
        values.put(AppDatabase.COL_PHOTO, photoPath); // null بتنخزّن NULL
        long id = helper.getWritableDatabase().insert(AppDatabase.TABLE_TREATED, null, values);
        if (id == -1) throw new SQLiteException("Insert into treated_cases failed");
        return id;
    }

    // الأحدث أولاً. كل كلمة لازم تظهر بالوصف أو الإجراءات أو التاريخ أو اسم الحالة
    // (AND بين الكلمات، OR بين الأعمدة). اسم الحالة وخطورتها بيجوا من JOIN مع emergencies
    public List<TreatedCase> search(@Nullable String rawQuery) {
        final String t = AppDatabase.TABLE_TREATED;
        final String e = AppDatabase.TABLE_EMERGENCIES;

        StringBuilder where = new StringBuilder();
        List<String> args = new ArrayList<>();
        String query = rawQuery == null ? "" : rawQuery.trim();
        for (String token : query.split("\\s+")) {
            if (token.isEmpty()) continue;
            String normalized = TextNormalizer.normalize(token);
            if (normalized.isEmpty()) normalized = token;

            where.append(where.length() == 0 ? " WHERE " : " AND ");
            where.append("(")
                    .append(t).append('.').append(AppDatabase.COL_DESCRIPTION).append(" LIKE ? ESCAPE '\\' OR ")
                    .append(t).append('.').append(AppDatabase.COL_PROCEDURES).append(" LIKE ? ESCAPE '\\' OR ")
                    .append(t).append('.').append(AppDatabase.COL_TREATED_DATE).append(" LIKE ? ESCAPE '\\' OR ")
                    .append(e).append('.').append(AppDatabase.COL_SEARCH).append(" LIKE ? ESCAPE '\\' OR ")
                    .append(e).append('.').append(AppDatabase.COL_SEARCH_EN).append(" LIKE ? ESCAPE '\\')");
            String raw = "%" + TextNormalizer.escapeLike(token) + "%";
            String norm = "%" + TextNormalizer.escapeLike(normalized) + "%";
            args.add(raw);
            args.add(raw);
            args.add(raw);
            args.add(norm);
            args.add(norm);
        }

        String sql = "SELECT "
                + t + "." + AppDatabase.COL_TREATED_ID + ", "
                + t + "." + AppDatabase.COL_CASE_ID + ", "
                + Lang.pick(e + "." + AppDatabase.COL_TITLE, e + "." + AppDatabase.COL_TITLE_EN) + " AS case_title, "
                + e + "." + AppDatabase.COL_DANGER + ", "
                + t + "." + AppDatabase.COL_DESCRIPTION + ", "
                + t + "." + AppDatabase.COL_PROCEDURES + ", "
                + t + "." + AppDatabase.COL_TREATED_DATE + ", "
                + t + "." + AppDatabase.COL_PHOTO
                + " FROM " + t
                + " JOIN " + e + " ON " + e + "." + AppDatabase.COL_CASE_ID + " = " + t + "." + AppDatabase.COL_CASE_ID
                + where
                + " ORDER BY " + t + "." + AppDatabase.COL_TREATED_DATE + " DESC, "
                + t + "." + AppDatabase.COL_TREATED_ID + " DESC";

        List<TreatedCase> results = new ArrayList<>();
        SQLiteDatabase db = helper.getReadableDatabase();
        try (Cursor c = db.rawQuery(sql, args.isEmpty() ? null : args.toArray(new String[0]))) {
            int idCol = c.getColumnIndexOrThrow(AppDatabase.COL_TREATED_ID);
            int caseIdCol = c.getColumnIndexOrThrow(AppDatabase.COL_CASE_ID);
            int titleCol = c.getColumnIndexOrThrow("case_title");
            int dangerCol = c.getColumnIndexOrThrow(AppDatabase.COL_DANGER);
            int descCol = c.getColumnIndexOrThrow(AppDatabase.COL_DESCRIPTION);
            int procCol = c.getColumnIndexOrThrow(AppDatabase.COL_PROCEDURES);
            int dateCol = c.getColumnIndexOrThrow(AppDatabase.COL_TREATED_DATE);
            int photoCol = c.getColumnIndexOrThrow(AppDatabase.COL_PHOTO);

            while (c.moveToNext()) {
                results.add(new TreatedCase(
                        c.getLong(idCol),
                        c.getLong(caseIdCol),
                        c.getString(titleCol),
                        Severity.fromLevel(c.getInt(dangerCol)),
                        c.getString(descCol),
                        c.getString(procCol),
                        c.getString(dateCol),
                        c.getString(photoCol))); // null إذا NULL
            }
        }
        return results;
    }

    // بتحذف الصف وبترجّع مسار صورته (إن وُجد) حتى يحذف المستدعي الملف
    @Nullable
    public String delete(long id) {
        SQLiteDatabase db = helper.getWritableDatabase();
        String[] idArg = {String.valueOf(id)};
        String photo = null;
        try (Cursor c = db.query(AppDatabase.TABLE_TREATED,
                new String[]{AppDatabase.COL_PHOTO},
                AppDatabase.COL_TREATED_ID + " = ?", idArg, null, null, null)) {
            if (c.moveToFirst()) photo = c.getString(0);
        }
        db.delete(AppDatabase.TABLE_TREATED, AppDatabase.COL_TREATED_ID + " = ?", idArg);
        return photo;
    }
}

public class AddTreatedCaseActivity extends AppCompatActivity {

    private static final String STATE_CASE_ID = "state_case_id";
    private static final String STATE_DATE = "state_date";
    private static final String STATE_PHOTO = "state_photo";
    private static final String STATE_CAMERA_FILE = "state_camera_file";

    // خيط خلفي واحد: معالجة الصورة والكتابة بالقاعدة ما بتنفّذ على الـ Main Thread
    private final ExecutorService executor = Executors.newSingleThreadExecutor();

    private MaterialAutoCompleteTextView caseInput;
    private View caseSeverityRow;
    private ImageView caseSeverityIcon;
    private TextView caseSeverityText;

    private TextInputLayout proceduresLayout;
    private TextInputLayout dateLayout;
    private TextInputEditText descriptionInput;
    private TextInputEditText proceduresInput;
    private TextInputEditText dateInput;
    private ShapeableImageView photoPreview;
    private MaterialButton addPhotoButton;
    private MaterialButton removePhotoButton;
    private MaterialButton saveButton;

    private List<CaseOption> caseOptions;

    private LocalDate selectedDate = LocalDate.now();
    @Nullable private Uri photoUri;
    @Nullable private File cameraFile;

    // لازم تتسجّل قبل أن تبدأ الـ Activity، لهيك هي حقول وليست داخل onCreate
    private final ActivityResultLauncher<PickVisualMediaRequest> galleryLauncher =
            registerForActivityResult(new ActivityResultContracts.PickVisualMedia(), uri -> {

    private final ActivityResultLauncher<Uri> cameraLauncher =
            registerForActivityResult(new ActivityResultContracts.TakePicture(), taken -> {
                if (taken && cameraFile != null) {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_add_treated_case);

        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.addRoot), (v, insets) -> {

            Insets ime = insets.getInsets(WindowInsetsCompat.Type.ime());
            // زر الحفظ بيرتفع فوق الكيبورد لما يفتح

        dateLayout.setEndIconOnClickListener(v -> showDatePicker());

    protected void onSaveInstanceState(@NonNull Bundle outState) {
        super.onSaveInstanceState(outState);
        outState.putLong(STATE_CASE_ID, selectedCaseId);
        outState.putString(STATE_DATE, selectedDate.toString());
        if (photoUri != null) outState.putString(STATE_PHOTO, photoUri.toString());
        if (cameraFile != null) outState.putString(STATE_CAMERA_FILE, cameraFile.getPath());

        super.onDestroy();
        executor.shutdown(); // بيخلي المهمة الجارية تكمل ولا بيقبل جديدة

    private void restoreState(@Nullable Bundle state) {
        if (state == null) return;
        selectedCaseId = state.getLong(STATE_CASE_ID, -1L);
        String date = state.getString(STATE_DATE);
        if (date != null) {
            try {
                selectedDate = LocalDate.parse(date);
            } catch (DateTimeParseException ignored) {
                // قيمة تالفة: نضل على تاريخ اليوم
            }
        }
        String photo = state.getString(STATE_PHOTO);
        if (photo != null) photoUri = Uri.parse(photo);
        String camera = state.getString(STATE_CAMERA_FILE);
        if (camera != null) cameraFile = new File(camera);

    // ---------- اختيار الحالة (من قائمة الدليل فقط، بدون كتابة) ----------

    private void setupCaseDropdown() {
        caseOptions = new EmergencyRepository(this).getCaseOptions();
        caseInput.setAdapter(new CaseAdapter(this, caseOptions));
        caseInput.setOnItemClickListener((parent, view, position, rowId) -> {
            caseLayout.setError(null);
            showCase(caseOptions.get(position));
        });
        // بعد تدوير الشاشة أو تغيير اللغة: نرجّع الحالة المختارة (بالرقم، فاسمها بيتجدد بلغة الجهاز)
        for (CaseOption option : caseOptions) {
            if (option.getId() == selectedCaseId) {
                showCase(option);
                break;
            }
        }
    }

    private void showCase(CaseOption option) {
        selectedCaseId = option.getId();
        caseInput.setText(option.getTitle(), false); // false = بدون فلترة القائمة
        caseSeverityIcon.setImageResource(option.getSeverity().iconRes);
        caseSeverityText.setText(option.getSeverity().labelRes);

    private static class CaseAdapter extends ArrayAdapter<CaseOption> {
        private final List<CaseOption> all;

        CaseAdapter(Context context, List<CaseOption> items) {
            super(context, android.R.layout.simple_dropdown_item_1line, items);
            this.all = items;
        }

        @NonNull
        @Override
        public Filter getFilter() {
            return new Filter() {
                @Override
                protected FilterResults performFiltering(CharSequence constraint) {
                    FilterResults results = new FilterResults();
                    results.values = all;
                    results.count = all.size();
                    return results;
                }

                @Override
                protected void publishResults(CharSequence constraint, FilterResults results) {
                    notifyDataSetChanged();
                }
            };
        }
    }

    // ---------- التاريخ ----------

    private void updateDateField() {
        dateInput.setText(selectedDate.format(DateTimeFormatter.ofLocalizedDate(FormatStyle.LONG)));
    }

    private void showDatePicker() {
        // MaterialDatePicker بيتعامل مع التواريخ كـ "منتصف الليل UTC"، فنحوّل بنفس المنطق
        long todayUtc = LocalDate.now().atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();
        long selectedUtc = selectedDate.atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();

        MaterialDatePicker<Long> picker = MaterialDatePicker.Builder.datePicker()
                .setTitleText(R.string.treated_date_title)
                .setSelection(selectedUtc)
                .setCalendarConstraints(new CalendarConstraints.Builder()
                        .setValidator(DateValidatorPointBackward.before(todayUtc)) // لا مستقبل
                        .build())
                .build();
        picker.addOnPositiveButtonClickListener(selection -> {
            selectedDate = Instant.ofEpochMilli(selection).atZone(ZoneOffset.UTC).toLocalDate();
            dateLayout.setError(null);
            updateDateField();
        });
        picker.show(getSupportFragmentManager(), "date_picker");
    }

    // ---------- الصورة ----------

    private void showPhotoSourceDialog() {
        new MaterialAlertDialogBuilder(this)
                .setTitle(R.string.treated_photo_source_title)
                .setItems(new CharSequence[]{
                        getString(R.string.photo_camera),
                        getString(R.string.photo_gallery)
                }, (dialog, which) -> {
                    if (which == 0) launchCamera();
                    else launchGallery();
                })
                .show();
    }

    private void launchGallery() {
        galleryLauncher.launch(new PickVisualMediaRequest.Builder()
                .setMediaType(ActivityResultContracts.PickVisualMedia.ImageOnly.INSTANCE)
                .build());
    }

    private void launchCamera() {
        try {
            File dir = new File(getCacheDir(), "camera");
            if (!dir.exists() && !dir.mkdirs()) throw new IOException("Cannot create " + dir);
            cameraFile = File.createTempFile("camera_", ".jpg", dir);
            cameraLauncher.launch(uriFor(cameraFile));
        } catch (IOException | ActivityNotFoundException e) {
            discardCameraFile();
            Toast.makeText(this, R.string.no_camera, Toast.LENGTH_SHORT).show();
        }
    }

    private Uri uriFor(File file) {
        return FileProvider.getUriForFile(this, getPackageName() + ".fileprovider", file);
    }

    private void discardCameraFile() {
        if (cameraFile != null) {
            //noinspection ResultOfMethodCallIgnored
            cameraFile.delete();
            cameraFile = null;
        }
    }

    private void setPhoto(Uri uri) {
        photoUri = uri;
        updatePhotoUi();
    }

    private void clearPhoto() {
        photoUri = null;
        discardCameraFile();
        updatePhotoUi();
    }

    private void updatePhotoUi() {
        boolean hasPhoto = photoUri != null;
        photoPreview.setVisibility(hasPhoto ? View.VISIBLE : View.GONE);
        removePhotoButton.setVisibility(hasPhoto ? View.VISIBLE : View.GONE);
        addPhotoButton.setText(hasPhoto ? R.string.treated_change_photo : R.string.treated_add_photo);
        if (hasPhoto) {
            Glide.with(this).load(photoUri).centerCrop().into(photoPreview);
        }
    }

    // ---------- الحفظ ----------

    private void save() {
        String description = textOf(descriptionInput);
        String procedures = textOf(proceduresInput);

        caseLayout.setError(selectedCaseId < 0
                ? getString(R.string.treated_case_required) : null);
        descriptionLayout.setError(description.isEmpty()
                ? getString(R.string.treated_description_required) : null);
        proceduresLayout.setError(procedures.isEmpty()
                ? getString(R.string.treated_procedures_required) : null);
        dateLayout.setError(selectedDate.isAfter(LocalDate.now())
                ? getString(R.string.treated_date_future) : null);

        if (caseLayout.getError() != null
                || descriptionLayout.getError() != null
                || proceduresLayout.getError() != null
                || dateLayout.getError() != null) {
            return; // الشاشة تضل مفتوحة وتعرض الأخطاء
        }

        saveButton.setEnabled(false); // يمنع الضغط المزدوج
        final Uri photo = photoUri;
        final String dateIso = selectedDate.toString();
        final long caseId = selectedCaseId;

        executor.execute(() -> {
            String relativePath = null;
            try {
                if (photo != null) {
                    relativePath = PhotoStorage.saveScaledCopy(getApplicationContext(), photo);
                }
                new TreatedCaseRepository(getApplicationContext())
                        .insert(caseId, description, procedures, dateIso, relativePath);
            } catch (IOException | RuntimeException e) {
                // فشل أي جزء: نحذف الصورة اللي انحفظت (لتجنب ملف بدون صف) ونرجّع الزر
                PhotoStorage.delete(getApplicationContext(), relativePath);
                runOnUiThread(() -> {
                    saveButton.setEnabled(true);
                    Toast.makeText(this, R.string.treated_save_failed, Toast.LENGTH_LONG).show();
                });
                return;
            }
            discardCameraFile();
            runOnUiThread(() -> {
                setResult(RESULT_OK);
                finish();
            });
        });
    }

    private static String textOf(TextInputEditText input) {
        return input.getText() == null ? "" : input.getText().toString().trim();
    }
}

package com.ahmed.firstaidhero1.ui;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.widget.TextView;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.appcompat.app.AlertDialog;
import androidx.fragment.app.Fragment;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import com.ahmed.firstaidhero1.R;
import com.ahmed.firstaidhero1.adapter.ContactAdapter;
import com.ahmed.firstaidhero1.data.ContactRepository;
import com.ahmed.firstaidhero1.model.Contact;
import com.ahmed.firstaidhero1.util.DialHelper;
import com.ahmed.firstaidhero1.util.PhoneValidator;
import com.google.android.material.dialog.MaterialAlertDialogBuilder;
import com.google.android.material.textfield.TextInputEditText;
import com.google.android.material.textfield.TextInputLayout;

import java.util.List;

public class ContactsFragment extends Fragment {

    private ContactRepository repository;
    private ContactAdapter adapter;
    private TextView emptyHint;

    public ContactsFragment() {
        super(R.layout.fragment_contacts);
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        repository = new ContactRepository(requireContext());
        emptyHint = view.findViewById(R.id.contactsEmpty);

        adapter = new ContactAdapter(new ContactAdapter.Listener() {
            @Override
            public void onCall(Contact contact) {
                DialHelper.dial(requireContext(), contact.getPhone());
            }

            @Override
            public void onEdit(Contact contact) {
                showContactDialog(contact);
            }

            @Override
            public void onDelete(Contact contact) {
                confirmDelete(contact);
            }
        });

        RecyclerView recyclerView = view.findViewById(R.id.contactsRecycler);
        recyclerView.setLayoutManager(new LinearLayoutManager(requireContext()));
        recyclerView.setAdapter(adapter);

        view.findViewById(R.id.addContactFab).setOnClickListener(v -> showContactDialog(null));

        reload();
    }

    private void reload() {
        List<Contact> contacts = repository.getAll();
        adapter.setItems(contacts);

        boolean hasUserContacts = false;
        for (Contact c : contacts) {
            if (!c.isBuiltIn()) {
                hasUserContacts = true;
                break;
            }
        }
        emptyHint.setVisibility(hasUserContacts ? View.GONE : View.VISIBLE);
    }

    // نفس الحوار للإضافة (existing = null) وللتعديل (existing = الرقم الحالي)
    private void showContactDialog(@Nullable Contact existing) {
        View form = LayoutInflater.from(requireContext())
                .inflate(R.layout.dialog_add_contact, null);
        TextInputLayout nameLayout = form.findViewById(R.id.nameLayout);
        TextInputEditText nameInput = form.findViewById(R.id.nameInput);
        TextInputLayout phoneLayout = form.findViewById(R.id.phoneLayout);
        TextInputEditText phoneInput = form.findViewById(R.id.phoneInput);

        if (existing != null) {
            nameInput.setText(existing.getName());
            phoneInput.setText(existing.getPhone());
        }

        AlertDialog dialog = new MaterialAlertDialogBuilder(requireContext())
                .setTitle(existing == null ? R.string.contact_add_title : R.string.contact_edit_title)
                .setView(form)
                .setPositiveButton(R.string.save, null)
                .setNegativeButton(R.string.cancel, null)
                .create();
        dialog.show();

        dialog.getButton(AlertDialog.BUTTON_POSITIVE).setOnClickListener(v -> {
            String name = textOf(nameInput).replaceAll("\\s+", " ");
            String phone = textOf(phoneInput);
            long editingId = existing == null ? -1 : existing.getId();

            nameLayout.setError(validateName(name));
            phoneLayout.setError(validatePhone(phone, editingId));
            if (nameLayout.getError() != null || phoneLayout.getError() != null) {
                return; // الحوار يضل مفتوح ويعرض الأخطاء
            }

            if (existing == null) {
                repository.insert(name, phone);
            } else {
                repository.update(existing.getId(), name, phone);
            }
            dialog.dismiss();
            reload();
        });
    }

    private void confirmDelete(Contact contact) {
        new MaterialAlertDialogBuilder(requireContext())
                .setTitle(R.string.contact_delete_title)
                .setMessage(getString(R.string.contact_delete_message, contact.getName()))
                .setPositiveButton(R.string.delete, (d, which) -> {
                    repository.delete(contact.getId());
                    reload();
                })
                .setNegativeButton(R.string.cancel, null)
                .show();
    }

    // كل دالة تحقق بترجع نص الخطأ، أو null إذا المدخل سليم
    @Nullable
    private String validateName(String name) {
        return name.isEmpty() ? getString(R.string.contact_name_required) : null;
    }

    @Nullable
    private String validatePhone(String phone, long excludeId) {
        if (!PhoneValidator.isValid(phone)) return getString(R.string.contact_phone_invalid);
        if (repository.phoneExists(phone, excludeId)) return getString(R.string.contact_phone_duplicate);
        return null;
    }

    private static String textOf(TextInputEditText input) {
        return input.getText() == null ? "" : input.getText().toString().trim();
    }
}

                showTab(TAG_CONTACTS);
            } else if (id == R.id.nav_treated) {
                showTab(TAG_TREATED);
            } else if (id == R.id.nav_stats) {
                showTab(TAG_STATS);
            } else {
                return false;
            }
            return true;
        });
    }

    private void setupFragments(Bundle savedInstanceState) {
        if (savedInstanceState != null) return;

        Fragment contacts = new ContactsFragment();
        Fragment treated = new TreatedCasesFragment();
        Fragment stats = new StatisticsFragment();

        getSupportFragmentManager().beginTransaction()
                .setReorderingAllowed(true)
                .add(R.id.fragmentContainer, new HomeFragment(), TAG_HOME)
                .add(R.id.fragmentContainer, contacts, TAG_CONTACTS).hide(contacts)
                .add(R.id.fragmentContainer, treated, TAG_TREATED).hide(treated)
                .add(R.id.fragmentContainer, stats, TAG_STATS).hide(stats)
                .commit();
    }

    private void showTab(String selectedTag) {
        FragmentManager fm = getSupportFragmentManager();
        FragmentTransaction ft = fm.beginTransaction().setReorderingAllowed(true);
        for (String tag : TAGS) {
            Fragment f = fm.findFragmentByTag(tag);
            if (f == null) continue;
            if (tag.equals(selectedTag)) ft.show(f);
            else ft.hide(f);
        }
        ft.commit();
    }
}

    private void loadStats() {

        int total = repository.getTreatedTotal();

        boolean empty = total == 0;
        emptyView.setVisibility(empty ? View.VISIBLE : View.GONE);

        for (SeverityStat stat : repository.getSeverityStats()) {

                    stat.getSeverity().iconRes,
                    getString(stat.getSeverity().nameRes),
                    stat.getCount(), total,

        }

        List<CaseStat> cases = repository.getCaseStats();
        int maxCase = 0;
        for (CaseStat stat : cases) maxCase = Math.max(maxCase, stat.getCount());

                    stat.getSeverity().iconRes,
                    stat.getTitle(),

package com.ahmed.firstaidhero1.util;

import java.util.regex.Pattern;

public final class PhoneValidator {

    // + اختيارية وبالبداية فقط، بعدها أرقام ومسافات وشرطات وأقواس
    private static final Pattern ALLOWED = Pattern.compile("^\\+?[0-9\\s\\-()]{3,20}$");

    private PhoneValidator() {}

    public static boolean isValid(String phone) {
        if (phone == null || !ALLOWED.matcher(phone).matches()) return false;
        int digits = canonical(phone).replace("+", "").length();
        return digits >= 3 && digits <= 15; // 15 = أقصى طول لرقم دولي (معيار E.164)
    }

    // الرقم بدون مسافات وشرطات وأقواس: للمقارنة واكتشاف التكرار
    public static String canonical(String phone) {
        return phone.replaceAll("[\\s\\-()]", "");
    }
}

===== app\src\main\java\com\ahmed\firstaidhero1\util\PhotoStorage.java =====
package com.ahmed.firstaidhero1.util;

import android.content.ContentResolver;
import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Matrix;
import android.media.ExifInterface;
import android.net.Uri;

import androidx.annotation.Nullable;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.util.UUID;

public final class PhotoStorage {

    private static final String DIR = "photos";
    private static final int MAX_SIDE_PX = 1280;
    private static final int JPEG_QUALITY = 85;

    private PhotoStorage() {}

    public static File resolve(Context context, String relativePath) {
        return new File(context.getFilesDir(), relativePath);
    }

    public static void delete(Context context, @Nullable String relativePath) {
        if (relativePath == null) return;
        File file = resolve(context, relativePath);
        if (file.exists() && !file.delete()) file.deleteOnExit();
    }

    // بتقرأ الصورة من أي Uri، بتصغّرها وبتصحح اتجاهها، وبتحفظها JPEG. بترجّع المسار النسبي
    public static String saveScaledCopy(Context context, Uri source) throws IOException {
        ContentResolver resolver = context.getContentResolver();

        // 1) قياس الأبعاد فقط، بدون تحميل الصورة بالذاكرة
        BitmapFactory.Options bounds = new BitmapFactory.Options();
        bounds.inJustDecodeBounds = true;
        try (InputStream in = open(resolver, source)) {
            BitmapFactory.decodeStream(in, null, bounds);
        }
        if (bounds.outWidth <= 0 || bounds.outHeight <= 0) {
            throw new IOException("Not a valid image: " + source);
        }

        // 2) تحميل نسخة مصغّرة بنسبة قوة 2 (أسرع وأوفر بالذاكرة)
        BitmapFactory.Options options = new BitmapFactory.Options();
        options.inSampleSize = sampleSize(bounds.outWidth, bounds.outHeight);
        Bitmap decoded;
        try (InputStream in = open(resolver, source)) {
            decoded = BitmapFactory.decodeStream(in, null, options);
        }
        if (decoded == null) throw new IOException("Could not decode image");

        // 3) تدوير حسب EXIF + تصغير للحجم النهائي بخطوة وحدة
        int rotation = readRotation(resolver, source);
        float scale = Math.min(1f, (float) MAX_SIDE_PX / Math.max(decoded.getWidth(), decoded.getHeight()));
        Bitmap result = decoded;
        if (rotation != 0 || scale < 1f) {
            Matrix matrix = new Matrix();
            matrix.postRotate(rotation);
            matrix.postScale(scale, scale);
            result = Bitmap.createBitmap(decoded, 0, 0, decoded.getWidth(), decoded.getHeight(), matrix, true);
            if (result != decoded) decoded.recycle();
        }

        // 4) حفظ JPEG بمجلد التطبيق الخاص
        File dir = new File(context.getFilesDir(), DIR);
        if (!dir.exists() && !dir.mkdirs()) throw new IOException("Cannot create " + dir);
        String relative = DIR + "/" + UUID.randomUUID() + ".jpg";
        File dest = resolve(context, relative);
        try (OutputStream out = new FileOutputStream(dest)) {
            if (!result.compress(Bitmap.CompressFormat.JPEG, JPEG_QUALITY, out)) {
                throw new IOException("Compress failed");
            }
        } catch (IOException e) {
            dest.delete();
            throw e;
        } finally {
            result.recycle();
        }
        return relative;
    }

    private static InputStream open(ContentResolver resolver, Uri uri) throws IOException {
        InputStream in = resolver.openInputStream(uri);
        if (in == null) throw new IOException("Cannot open " + uri);
        return in;
    }

    // أكبر قوة 2 بتخلي الأبعاد لسا >= الحجم المطلوب
    private static int sampleSize(int width, int height) {
        int sample = 1;
        while (Math.max(width, height) / (sample * 2) >= MAX_SIDE_PX) {
            sample *= 2;
        }
        return sample;
    }

    private static int readRotation(ContentResolver resolver, Uri uri) {
        try (InputStream in = resolver.openInputStream(uri)) {
            if (in == null) return 0;
            ExifInterface exif = new ExifInterface(in);
            switch (exif.getAttributeInt(ExifInterface.TAG_ORIENTATION,
                    ExifInterface.ORIENTATION_NORMAL)) {
                case ExifInterface.ORIENTATION_ROTATE_90:
                    return 90;
                case ExifInterface.ORIENTATION_ROTATE_180:
                    return 180;
                case ExifInterface.ORIENTATION_ROTATE_270:
                    return 270;
                default:
                    return 0;
            }
        } catch (IOException e) {
            return 0; // ما قدرنا نقرأ EXIF: نعتبر الصورة معتدلة
        }
    }
}

package com.ahmed.firstaidhero1.util;

import java.util.Locale;

public final class TextNormalizer {

    private TextNormalizer() {}

    // لازم تطابق دالة normalize بملف build_db.py بالضبط، وإلا ما بيتطابق البحث
    public static String normalize(String input) {
        if (input == null) return "";
        StringBuilder sb = new StringBuilder(input.length());
        for (char c : input.toLowerCase(Locale.ROOT).toCharArray()) {
            if ((c >= '\u064B' && c <= '\u0652') || c == '\u0640') continue; // تشكيل وتطويل
            switch (c) {
                case '\u0623': // أ
                case '\u0625': // إ
                case '\u0622': // آ
                    sb.append('\u0627'); // ا
                    break;
                case '\u0649': // ى
                    sb.append('\u064A'); // ي
                    break;
                case '\u0629': // ة
                    sb.append('\u0647'); // ه
                    break;
                default:
                    sb.append(c);
            }
        }
        return sb.toString().trim().replaceAll("\\s+", " ");
    }

    // بيعطّل الرموز الخاصة بـ LIKE (% و _) حتى تُعامل كحروف عادية
    public static String escapeLike(String token) {
        return token.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_");
    }
}
